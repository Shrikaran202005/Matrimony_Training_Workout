# Weather Dashboard API - cURL Test Commands (PowerShell)

# Base URL: http://localhost:3000

# 1. Health check
curl.exe http://localhost:3000/health

# 2. Feature 1 & 2: Multiple cities + formattedDate
curl.exe "http://localhost:3000/weather?cities=London,Paris,Tokyo"

curl.exe "http://localhost:3000/forecast/London"

curl.exe "http://localhost:3000/weather/compare?cities=London,Moscow"
